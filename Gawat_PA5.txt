%% Problem B: TWO METHODS FOR A LINEAR SYSTEM

% Define the coefficient matrix A and constant vector b
A = [ 2 -1  3  2;
      1  4 -2  1;
      3  2  1 -1;
     -2  3  4  2];

b = [16; -7; 6; 7];

% 1. Inverse Method
x_inv = inv(A) * b

% 2. Cramer's Rule
n = length(b);
x_cramer = zeros(n, 1);
det_A = det(A);

for i = 1:n
    A_temp = A;
    A_temp(:, i) = b; % Replace the i-th column of A with b
    x_cramer(i) = det(A_temp) / det_A;
end

x_cramer

% Required Numerical Checks
det_A_check = det(A)
diff_norm = norm(x_inv - x_cramer)
residual_norm = norm(A * x_inv - b)
